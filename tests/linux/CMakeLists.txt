cmake_minimum_required(VERSION 3.16)
project(kirikinux2_components LANGUAGES C CXX)
get_filename_component(KR2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(KR2_THIRD_PARTY "${KR2_ROOT}/third_party" CACHE PATH "Downloaded archive sources")
option(KR2_COMPONENT_ARCHIVES "Compile and link the downloaded 7-Zip/UnRAR targets" OFF)
option(KR2_COMPONENT_SANITIZERS "Check container fixes with ASan and UBSan" ON)
set(KR2_WITH_JXR OFF)
include("${KR2_ROOT}/cmake/Kr2Sources.cmake")
foreach(required visual/WindowIntf.cpp visual/LayerIntf.cpp environ/cocos2d/MainScene.cpp environ/cocos2d/AppDelegate.cpp environ/ConfigManager/GlobalConfigManager.cpp)
  if(NOT "${KR2_ROOT}/src/core/${required}" IN_LIST KR2_CORE_SOURCES)
    message(FATAL_ERROR "Engine source missing from main target: ${required}")
  endif()
endforeach()
foreach(source IN LISTS KR2_CORE_SOURCES KR2_PLUGIN_SOURCES KR2_PLATFORM_SOURCES)
  if(NOT EXISTS "${source}")
    message(FATAL_ERROR "Source manifest names nonexistent file: ${source}")
  endif()
endforeach()
add_executable(port_components port_components.cpp)
target_compile_features(port_components PRIVATE cxx_std_14)
target_include_directories(port_components PRIVATE "${KR2_ROOT}/src/core/tjs2" "${KR2_ROOT}/src/core/environ" "${KR2_ROOT}/platform/linux/compat")
if(KR2_COMPONENT_SANITIZERS)
  target_compile_options(port_components PRIVATE -g -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(port_components PRIVATE -fsanitize=address,undefined)
endif()
add_executable(tvpgl_components tvpgl_components.cpp "${KR2_ROOT}/src/core/visual/tvpgl.cpp" "${KR2_ROOT}/src/core/visual/gl/blend_function.cpp")
target_compile_features(tvpgl_components PRIVATE cxx_std_14)
target_include_directories(tvpgl_components PRIVATE "${KR2_ROOT}/src/core/tjs2" "${KR2_ROOT}/src/core/visual" "${KR2_ROOT}/src/core/visual/gl" "${KR2_ROOT}/platform/linux/compat")
add_library(linux_platform_fallbacks OBJECT "${KR2_ROOT}/platform/linux/LinuxDialogs.cpp" "${KR2_ROOT}/platform/linux/LinuxNoNeonStubs.c")
target_compile_features(linux_platform_fallbacks PRIVATE cxx_std_14)
# Compile and link the real GTK branch as well as the no-GTK fallback, so GTK
# include-ordering problems are caught here.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(KR2_COMPONENT_GTK3 QUIET IMPORTED_TARGET gtk+-3.0)
endif()
if(KR2_COMPONENT_GTK3_FOUND)
  pkg_check_modules(KR2_COMPONENT_FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
  add_executable(gtk_dialog_components gtk_dialog_components.cpp "${KR2_ROOT}/platform/linux/LinuxDialogs.cpp")
  target_compile_features(gtk_dialog_components PRIVATE cxx_std_14)
  target_compile_definitions(gtk_dialog_components PRIVATE KR2_LINUX_HAVE_GTK3=1)
  target_link_libraries(gtk_dialog_components PRIVATE PkgConfig::KR2_COMPONENT_GTK3 PkgConfig::KR2_COMPONENT_FONTCONFIG)
  add_executable(desktop_dialogs desktop_dialogs.cpp "${KR2_ROOT}/platform/linux/LinuxDialogs.cpp")
  target_compile_features(desktop_dialogs PRIVATE cxx_std_14)
  target_compile_definitions(desktop_dialogs PRIVATE KR2_LINUX_HAVE_GTK3=1)
  target_link_libraries(desktop_dialogs PRIVATE PkgConfig::KR2_COMPONENT_GTK3 PkgConfig::KR2_COMPONENT_FONTCONFIG)
else()
  message(STATUS "GTK development files unavailable: real GTK dialog check skipped")
endif()
enable_testing()
add_test(NAME portable-container-and-allocator COMMAND port_components)
add_test(NAME tvpgl-265-entrypoints COMMAND tvpgl_components)
if(TARGET gtk_dialog_components)
  add_test(NAME gtk-dialog-headless-fallback COMMAND gtk_dialog_components)
  set_tests_properties(gtk-dialog-headless-fallback PROPERTIES ENVIRONMENT "DISPLAY=;WAYLAND_DISPLAY=;GDK_BACKEND=x11")
endif()

# Windows face names from the OpenType name table (FontNameTable.cpp).
# Run with a font, e.g.:
#   font_name_table SourceHanSansCN-Bold.otf 0 zh_CN "Source Han Sans CN Bold"
if(PkgConfig_FOUND)
  pkg_check_modules(KR2_COMPONENT_FREETYPE QUIET IMPORTED_TARGET freetype2)
endif()
if(KR2_COMPONENT_FREETYPE_FOUND)
  add_executable(font_name_table font_name_table.cpp "${KR2_ROOT}/src/core/visual/FontNameTable.cpp")
  target_compile_features(font_name_table PRIVATE cxx_std_14)
  target_include_directories(font_name_table PRIVATE "${KR2_ROOT}/src/core/visual")
  target_link_libraries(font_name_table PRIVATE PkgConfig::KR2_COMPONENT_FREETYPE)
  foreach(font /usr/share/fonts/opentype/noto/NotoSansCJK-Bold.ttc)
    if(EXISTS "${font}")
      add_test(NAME font-name-table-noto-cjk COMMAND font_name_table "${font}" 2 ja_JP "Noto Sans CJK SC" "Noto Sans CJK SC Bold")
    endif()
  endforeach()
endif()
if(KR2_COMPONENT_ARCHIVES)
  set(KR2_FORCE_BUNDLED_UNRAR ON)
  include("${KR2_ROOT}/cmake/Kr2ArchiveCodecs.cmake")
  add_executable(archive_components archive_components.cpp)
  target_link_libraries(archive_components PRIVATE kr2_7zip kr2_unrar)
  add_test(NAME archive-codec-link COMMAND archive_components)
endif()
